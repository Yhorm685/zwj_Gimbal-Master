cmake_minimum_required(VERSION 3.22)
project(gimbal_checks LANGUAGES CXX)
enable_testing()
add_library(gimbal_core ../Module/gimbal.cpp)
target_compile_features(gimbal_core PUBLIC cxx_std_17)
target_include_directories(gimbal_core PUBLIC ..)
add_executable(control_test control_test.cpp)
target_link_libraries(control_test PRIVATE gimbal_core)
add_test(NAME control COMMAND control_test)
add_executable(protocol_test protocol_test.cpp ../Application/protocol.cpp)
target_link_libraries(protocol_test PRIVATE gimbal_core)
add_test(NAME serial_protocol COMMAND protocol_test)
add_executable(uart_test uart_test.cpp ../BSP/uart.cpp)
target_compile_features(uart_test PRIVATE cxx_std_17)
target_include_directories(uart_test PRIVATE support ..)
add_test(NAME dual_uart COMMAND uart_test)
